      ("NetIncome", "latest_annual"),
      ("NetIncome", "latest_quarter"),
      ("NetIncome", "latest_reported_quarter"),
      ("TotalAssets", "latest_quarter"),
  ]

  async def main():
      async with Client(URL) as client:
          for metric, period in QUESTIONS:
              args = {"cik": "320193", "metric": metric, "period": period}
              r = await client.call_tool("get_fact", args)
              if r.is_error:
                  print(f"{metric:<20} {period:<24} ERROR {r.content[0].text}")
                  continue
              a = r.structured_content
              print(f"{metric:<20} {period:<24} {a['cache']:<6} {a['value']:>16} {a['period']:<8} "
                    f"{a['start']} {a['end']} {a['form']} {a['accn']} {a['filed']}")
          s = (await client.call_tool("gateway_stats", {})).structured_content
      print("\nupstream requests to SEC:", s["upstream_attempts"])
      print(json.dumps(s["answers"], indent=2))

  asyncio.run(main())
