  import asyncio, json, time
  from collections import Counter
  from mcp import Client

  URL = "http://127.0.0.1:8000/mcp"
  QUESTIONS = [
      "Translate 'good morning' into French. Answer with the translation only.",
      "At sea level, what is the boiling point of water in Celsius? Answer with a number.",
      "Name the largest planet in the solar system. Answer with one word.",
      "In one sentence, what does a network router do?",
      "Write a haiku about waiting in line.",
  ]
  AGENTS = ["agent-1", "agent-2", "agent-3"]

  def ask(text, workload="support-team", agent="agent-1"):
      return {"messages": [{"role": "user", "content": text}], "max_tokens": 120,
              "workload": workload, "agent_id": agent}

  async def one(client, args):
      r = await client.call_tool("generate", args)
      if r.is_error:
          return "ERROR " + r.content[0].text, None
      a = r.structured_content
      return a["source"], a

  async def round_(client, label):
      start = time.perf_counter()
      calls = [one(client, ask(q, agent=g)) for q in QUESTIONS for g in AGENTS]
      results = await asyncio.gather(*calls)
      seconds = time.perf_counter() - start
      sources = Counter(source for source, _ in results)
      print(f"{label:<10} {len(results)} requests in {seconds:6.2f}s  {dict(sources)}")
      return results

  async def main():
      async with Client(URL) as client:
          first = await round_(client, "round 1")
          await round_(client, "round 2")
          source, _ = await one(client, ask(QUESTIONS[2], workload="team-b"))
          print(f"team-b     1 request  {source}")
          s = (await client.call_tool("gateway_stats", {})).structured_content
      print("\nanswers:")
      for i, q in enumerate(QUESTIONS):
          a = first[i * len(AGENTS)][1]
          if a:
              print(f"  {a['finish_reason']:<6} {a['prompt_tokens']}+{a['completion_tokens']:<4} "
                    f"{a['text'][:70]!r}")
      print("\ndata source configured:", "upstream_attempts" in s)
      print(json.dumps(s["inference"], indent=2))

  asyncio.run(main())
