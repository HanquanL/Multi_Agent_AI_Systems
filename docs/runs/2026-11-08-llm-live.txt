  import asyncio, json, time
  from mcp import Client

  URL = "http://127.0.0.1:8000/mcp"

  def ask(text, max_tokens=40, **extra):
      return {"messages": [{"role": "user", "content": text}], "max_tokens": max_tokens, **extra}

  PLAIN = ask("Name one primary color. Answer with one word.")
  TOGETHER = ask("Name one planet. Answer with one word.")
  GROUNDED = ask("What was Apple's FY2024 net income? Use the data given.", max_tokens=60,
                 context=["320193:NetIncome:FY2024"])
  CUT = ask("Count from one to one hundred in words.", max_tokens=3)

  async def one(client, label, args):
      start = time.perf_counter()
      r = await client.call_tool("generate", args)
      seconds = time.perf_counter() - start
      if r.is_error:
          print(f"{label:<14} ERROR {r.content[0].text}")
          return
      a = r.structured_content
      print(f"{label:<14} {a['source']:<9} {seconds:7.2f}s  {a['finish_reason']:<7} "
            f"tokens {a['prompt_tokens']}+{a['completion_tokens']}  {a['text'][:60]!r}")

  async def main():
      async with Client(URL) as client:
          await one(client, "plain", PLAIN)
          await one(client, "plain again", PLAIN)
          await one(client, "other team", {**PLAIN, "workload": "team-b"})
          await asyncio.gather(*(one(client, f"together {i}", TOGETHER) for i in range(3)))
          await one(client, "grounded", GROUNDED)
          await one(client, "grounded again", GROUNDED)
          await one(client, "cut off", CUT)
          await one(client, "cut off again", CUT)
          s = (await client.call_tool("gateway_stats", {})).structured_content
      print(json.dumps(s["inference"], indent=2))
      print("SEC requests:", s["upstream_attempts"])

  asyncio.run(main())
