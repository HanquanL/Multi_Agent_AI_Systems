Code:
  import asyncio
  from multi_agent_ai_systems.config import sec_user_agent
  from multi_agent_ai_systems.connectors.edgar import EdgarConnector, companyfacts_key, submissions_key

  CONCEPTS = ["RevenueFromContractWithCustomerExcludingAssessedTax", "OperatingIncomeLoss",
              "EarningsPerShareDiluted", "Assets", "CashAndCashEquivalentsAtCarryingValue"]

  async def main():
      async with EdgarConnector(sec_user_agent()) as conn:
          facts = (await conn.fetch(companyfacts_key("320193"))).data["facts"]["us-gaap"]
          recent = (await conn.fetch(submissions_key("320193"))).data["filings"]["recent"]
      for c in CONCEPTS:
          units = facts.get(c, {}).get("units", {})
          print(f"\n{c}: units={list(units)}")
          for unit, rows in units.items():
              for f in rows:
                  if f["end"].startswith("2024") or f["end"] == "2024-09-28":
                      print(" ", unit, f.get("start"), f["end"], f["fy"], f["fp"], f["form"], f["filed"], f["accn"], f["val"])
      print("\nsubmissions fields:", sorted(recent))
      for a, form, rd, fd in zip(recent["accessionNumber"], recent["form"], recent["reportDate"], recent["filingDate"]):
          if form in ("10-K", "10-Q") and fd.startswith(("2024", "2025")):
              print(" ", a, form, "report", rd, "filed", fd)

  asyncio.run(main())
